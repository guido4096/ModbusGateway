void handleDescription() {
    using namespace modbus_gateway;
    String result;
    result += getDeviceDescription<WattNode>();
    result += getDeviceDescription<EM24_E1>();
    result += getDeviceDescription<Alfen>();
    result += getDeviceDescription<SolarEdge>();

}
